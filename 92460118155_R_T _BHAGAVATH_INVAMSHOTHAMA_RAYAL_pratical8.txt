#include <iostream>
#include <vector>
#include <stack>
#include <queue>
#include <chrono>

using namespace std;
using namespace chrono;

class Graph
{
private:
    int vertices;
    vector<vector<int>> graph;

public:
    Graph(int v)
    {
        vertices = v;
        graph.resize(vertices);
    }

    void insertEdge(int source, int destination)
    {
        graph[source].push_back(destination);
        graph[destination].push_back(source);   // Remove for directed graph
    }

    void DFS(int start)
    {
        vector<bool> visited(vertices, false);
        stack<int> s;

        s.push(start);

        while (!s.empty())
        {
            int current = s.top();
            s.pop();

            if (!visited[current])
            {
                visited[current] = true;
                cout << current << " ";

                for (int i = graph[current].size() - 1; i >= 0; i--)
                {
                    int neighbour = graph[current][i];
                    if (!visited[neighbour])
                        s.push(neighbour);
                }
            }
        }
    }

    void BFS(int start)
    {
        vector<bool> visited(vertices, false);
        queue<int> q;

        visited[start] = true;
        q.push(start);

        while (!q.empty())
        {
            int current = q.front();
            q.pop();

            cout << current << " ";

            for (int neighbour : graph[current])
            {
                if (!visited[neighbour])
                {
                    visited[neighbour] = true;
                    q.push(neighbour);
                }
            }
        }
    }
};

int main()
{
    int V, E;

    cout << "Enter number of vertices: ";
    cin >> V;

    Graph g(V);

    cout << "Enter number of edges: ";
    cin >> E;

    cout << "Enter edges (source destination):" << endl;

    for (int i = 0; i < E; i++)
    {
        int u, v;
        cin >> u >> v;
        g.insertEdge(u, v);
    }

    int startVertex;
    cout << "Enter starting vertex: ";
    cin >> startVertex;

    auto dfsStart = high_resolution_clock::now();

    cout << "\nDFS Traversal: ";
    g.DFS(startVertex);

    auto dfsEnd = high_resolution_clock::now();

    auto bfsStart = high_resolution_clock::now();

    cout << "\n\nBFS Traversal: ";
    g.BFS(startVertex);

    auto bfsEnd = high_resolution_clock::now();

    auto dfsTime = duration_cast<nanoseconds>(dfsEnd - dfsStart);
    auto bfsTime = duration_cast<nanoseconds>(bfsEnd - bfsStart);

    cout << "\n\nExecution Time";
    cout << "\nDFS : " << dfsTime.count() << " ns";
    cout << "\nBFS : " << bfsTime.count() << " ns";

    return 0;
}
