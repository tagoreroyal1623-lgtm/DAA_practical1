#include <iostream>
using namespace std;

int main()
{
    int weight[] = {2, 1, 3, 2};
    int profit[] = {12, 10, 20, 15};

    int n = 4;
    int capacity = 5;

    int dp[5][6] = {0};

    // Build DP table
    for (int i = 1; i <= n; i++)
    {
        for (int w = 1; w <= capacity; w++)
        {
            if (weight[i - 1] <= w)
            {
                int take = profit[i - 1] + dp[i - 1][w - weight[i - 1]];
                int notTake = dp[i - 1][w];

                if (take > notTake)
                    dp[i][w] = take;
                else
                    dp[i][w] = notTake;
            }
            else
            {
                dp[i][w] = dp[i - 1][w];
            }
        }
    }

    // Find selected items
    int selected[4] = {0};

    int w = capacity;

    for (int i = n; i >= 1; i--)
    {
        if (dp[i][w] != dp[i - 1][w])
        {
            selected[i - 1] = 1;
            w = w - weight[i - 1];
        }
    }

    cout << "Selected Items = {";

    for (int i = 0; i < n; i++)
    {
        cout << selected[i];

        if (i < n - 1)
            cout << ",";
    }

    cout << "}" << endl;

    cout << "Maximum Profit = " << dp[n][capacity] << endl;

    return 0;
}
