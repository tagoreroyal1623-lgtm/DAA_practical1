#include <iostream>
#include <vector>
#include <chrono>

using namespace std;

// 1. Linear Search: Checks elements one by one from start to end
int linearSearch(const vector<int>& numbers, int target) {
    for (int i = 0; i < numbers.size(); i++) {
        if (numbers[i] == target) {
            return i; // Found, return index
        }
    }
    return -1; // Not found
}

// 2. Binary Search: Repeatedly divides the sorted list in half
int binarySearch(const vector<int>& numbers, int target) {
    int left = 0;
    int right = numbers.size() - 1;

    while (left <= right) {
        int middle = left + (right - left) / 2;

        if (numbers[middle] == target) {
            return middle; // Found, return index
        } 
        
        if (numbers[middle] < target) {
            left = middle + 1;  // Search in the right half
        } else {
            right = middle - 1; // Search in the left half
        }
    }
    return -1; // Not found
}

int main() {
    int totalElements = 100000;
    vector<int> numbers(totalElements);

    // Populate vector with numbers 1 to 100000
    for (int i = 0; i < totalElements; i++) {
        numbers[i] = i + 1;
    }

    int target;
    cout << "Enter element to search: ";
    cin >> target;

    // --- Linear Search Execution ---
    auto startLinear = chrono::high_resolution_clock::now();
    int linearResult = linearSearch(numbers, target);
    auto stopLinear = chrono::high_resolution_clock::now();

    cout << "\nLinear Search\n";
    if (linearResult != -1) {
        cout << "Element found at index " << linearResult << "\n";
    } else {
        cout << "Element not found\n";
    }
    
    auto linearTime = chrono::duration_cast<chrono::microseconds>(stopLinear - startLinear);
    cout << "Time Taken : " << linearTime.count() << " microseconds\n";


    // --- Binary Search Execution ---
    auto startBinary = chrono::high_resolution_clock::now();
    int binaryResult = binarySearch(numbers, target);
    auto stopBinary = chrono::high_resolution_clock::now();

    cout << "\nBinary Search\n";
    if (binaryResult != -1) {
        cout << "Element found at index " << binaryResult << "\n";
    } else {
        cout << "Element not found\n";
    }

    auto binaryTime = chrono::duration_cast<chrono::microseconds>(stopBinary - startBinary);
    cout << "Time Taken : " << binaryTime.count() << " microseconds\n";

    return 0;
}
