#include <iostream>
#include <chrono>

using namespace std;
using namespace std::chrono;

// Iterative Factorial Calculation
unsigned long long getFactorialLoop(int num) {
    unsigned long long fact = 1;
    int i = 1;
    while (i <= num) {
        fact *= i;
        i++;
    }
    return fact;
}

// Recursive Factorial Calculation
unsigned long long getFactorialRecursion(int num) {
    if (num == 0 || num == 1) {
        return 1;
    }
    return num * getFactorialRecursion(num - 1);
}

int main() {
    int n;
    cout << "Enter a non-negative integer (e.g., 20): ";
    
    cin >> n;
    if (cin.fail() || n < 0) {
        cerr << "Invalid input! Please enter a non-negative integer." << endl;
        return 1;
    }

    // Measure Iterative Time
    auto startTime1 = high_resolution_clock::now();
    unsigned long long iterativeAnswer = getFactorialLoop(n);
    auto endTime1 = high_resolution_clock::now();

    duration<double, nano> iterativeTime = endTime1 - startTime1;

    // Measure Recursive Time
    auto startTime2 = high_resolution_clock::now();
    unsigned long long recursiveAnswer = getFactorialRecursion(n);
    auto endTime2 = high_resolution_clock::now();

    duration<double, nano> recursiveTime = endTime2 - startTime2;

    // Display Output
    cout << "\n--- Results for " << n << "! ---\n";
    cout << "Iterative Result : " << iterativeAnswer << "\n";
    cout << "Iterative Time   : " << iterativeTime.count() << " ns\n";
    cout << "-------------------------------\n";
    cout << "Recursive Result : " << recursiveAnswer << "\n";
    cout << "Recursive Time   : " << recursiveTime.count() << " ns\n";

    return 0;
}
