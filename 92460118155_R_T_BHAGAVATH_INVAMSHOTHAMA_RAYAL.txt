#include <iostream>
using namespace std;

int main()
{
    int coins[] = {1, 2, 5};
    int n = 3;
    int amount = 11;

    int dp[3][12];

    // Initialize first row
    for (int j = 0; j <= amount; j++)
    {
        if (j % coins[0] == 0)
            dp[0][j] = j / coins[0];
        else
            dp[0][j] = 9999;
    }

    // Build DP table
    for (int i = 1; i < n; i++)
    {
        for (int j = 0; j <= amount; j++)
        {
            int notTake = dp[i - 1][j];

            int take = 9999;

            if (coins[i] <= j)
                take = 1 + dp[i][j - coins[i]];

            if (take < notTake)
                dp[i][j] = take;
            else
                dp[i][j] = notTake;
        }
    }

    // Find selected coins
    int selected[3] = {0};

    int j = amount;

    for (int i = n - 1; i >= 0; i--)
    {
        while (j >= coins[i] &&
               dp[i][j] == 1 + dp[i][j - coins[i]])
        {
            selected[i]++;
            j = j - coins[i];
        }
    }

    cout << "Selected Coins = {";

    for (int i = 0; i < n; i++)
    {
        cout << selected[i];

        if (i < n - 1)
            cout << ",";
    }

    cout << "}" << endl;

    cout << "Minimum Number of Coins = "
         << dp[n - 1][amount] << endl;

    return 0;
}
