#include <iostream>
using namespace std;

int main()
{
    int p[4];

    cout << "Enter dimensions of 3 matrices: ";
    cin >> p[0] >> p[1] >> p[2] >> p[3];

    int dp[4][4] = {0};

    dp[1][1] = 0;
    dp[2][2] = 0;
    dp[3][3] = 0;

    dp[1][2] = p[0] * p[1] * p[2];

    dp[2][3] = p[1] * p[2] * p[3];

    int cost1 = dp[1][2] + p[0] * p[2] * p[3];

    int cost2 = dp[2][3] + p[0] * p[1] * p[3];

    dp[1][3] = min(cost1, cost2);

    cout << "\nCost of (A1 x A2) x A3 = " << cost1;
    cout << "\nCost of A1 x (A2 x A3) = " << cost2;

    cout << "\n\nMinimum multiplication cost = "
         << dp[1][3] << endl;

    return 0;
}
