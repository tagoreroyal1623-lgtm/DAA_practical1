#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int n;

    cout << "Enter number of matrices: ";
    cin >> n;

    // n matrices need n+1 dimensions
    vector<int> p(n + 1);

    cout << "Enter " << n + 1 << " dimensions: ";

    for (int i = 0; i <= n; i++)
    {
        cin >> p[i];
    }

    // DP table
    vector<vector<int>> dp(n + 1, vector<int>(n + 1, 0));

    // length = number of matrices in the chain
    for (int length = 2; length <= n; length++)
    {
        // i = starting matrix
        for (int i = 1; i <= n - length + 1; i++)
        {
            // j = ending matrix
            int j = i + length - 1;

            // Assume a very large cost initially
            dp[i][j] = 999999999;

            // Try every possible splitting point
            for (int k = i; k < j; k++)
            {
                int cost = dp[i][k]
                         + dp[k + 1][j]
                         + p[i - 1] * p[k] * p[j];

                if (cost < dp[i][j])
                {
                    dp[i][j] = cost;
                }
            }
        }
    }

    cout << "\nMinimum multiplication cost = "
         << dp[1][n] << endl;

    return 0;
}
