#include <iostream>
using namespace std;

int main()
{
    int n, amount;

    cout << "Enter number of coins: ";
    cin >> n;

    int coin[n];

    cout << "Enter coin values: ";
    for (int i = 0; i < n; i++)
    {
        cin >> coin[i];
    }

    cout << "Enter amount: ";
    cin >> amount;
    int dp[amount + 1];

    for (int i = 0; i <= amount; i++)
    {
        dp[i] = 9999;
    }

    dp[0] = 0;

    for (int i = 1; i <= amount; i++)
    {
        for (int j = 0; j < n; j++)
        {
            if (coin[j] <= i)
            {
                dp[i] = min(dp[i],
                            dp[i - coin[j]] + 1);
            }
        }
    }

    if (dp[amount] == 9999)
        cout << "Amount cannot be made.";
    else
        cout << "Minimum number of coins = "
             << dp[amount];

    return 0;
}
